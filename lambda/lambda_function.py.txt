import io
import logging
import os
from pathlib import PurePosixPath
from urllib.parse import unquote_plus

import boto3
from PIL import Image, ImageOps, UnidentifiedImageError

s3 = boto3.client("s3")

logger = logging.getLogger()
logger.setLevel(logging.INFO)

ORIGINAL_BUCKET = os.environ["ORIGINAL_BUCKET"]
OUTPUT_BUCKET = os.environ["OUTPUT_BUCKET"]


def resize_image(bucket, key):
    # Only read images from our original image bucket.
    if bucket != ORIGINAL_BUCKET:
        raise ValueError("The source bucket is not allowed.")

    # Read the original image from S3.
    response = s3.get_object(Bucket=bucket, Key=key)

    try:
        image_bytes = response["Body"].read()
    finally:
        response["Body"].close()

    # Resize the image while keeping its original proportions.
    with Image.open(io.BytesIO(image_bytes)) as source:
        image = ImageOps.exif_transpose(source).convert("RGB")
        image.thumbnail((200, 200))

        output = io.BytesIO()
        image.save(output, format="JPEG", quality=85)

        width, height = image.size

    # Keep the source folder path to avoid filename collisions.
    output_key = (
        f"thumbnails/{PurePosixPath(key).with_suffix('.jpg')}"
    )

    # Save the thumbnail in the second bucket.
    s3.put_object(
        Bucket=OUTPUT_BUCKET,
        Key=output_key,
        Body=output.getvalue(),
        ContentType="image/jpeg"
    )

    logger.info("Created thumbnail: %s", output_key)

    return {
        "success": True,
        "output_bucket": OUTPUT_BUCKET,
        "output_key": output_key,
        "width": width,
        "height": height
    }


def lambda_handler(event, context):
    try:
        # Handle automatic uploads from the S3 trigger.
        if "Records" in event:
            results = []

            for record in event["Records"]:
                bucket = record["s3"]["bucket"]["name"]

                # S3 notification keys are URL encoded.
                key = unquote_plus(record["s3"]["object"]["key"])

                results.append(resize_image(bucket, key))

            return {
                "success": True,
                "results": results
            }

        # Handle requests received through Step Functions.
        return resize_image(event["bucket"], event["key"])

    except (UnidentifiedImageError, ValueError, KeyError) as error:
        logger.exception("Invalid image or request")

        # Let Step Functions inspect an expected processing failure.
        if "Records" not in event:
            return {
                "success": False,
                "error": str(error)
            }

        raise

    except Exception:
        # Raise unexpected errors so AWS can report or retry them.
        logger.exception("Image processing failed")
        raise
